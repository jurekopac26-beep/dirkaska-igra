phase('Pregled')
const reviews = await parallel(LENSES.map(l => () =>
  agent(COMMON + '\nYour key: ' + l.key + '\n\n' + l.prompt, { label: 'pregled:' + l.key, phase: 'Pregled', schema: FINDINGS })
    .then(r => r ? { key: l.key, summary: r.summary, findings: r.findings.map(f => ({ ...f, lens: l.key })) } : null)
))
const ok = reviews.filter(Boolean)
const all = ok.flatMap(r => r.findings)
log('Najdb: ' + all.length + ' (' + ok.map(r => r.key + ' ' + r.findings.length).join(', ') + ')')

// Preverjanje: high in medium najdbe po skupinah (največ 4 skeptiki, da tok ostane pri 10 agentih); low gredo naprej nepreverjene
const serious = all.filter(f => f.severity !== 'low')
const low = all.filter(f => f.severity === 'low')
if (low.length) log('Nizkih najdb brez preverjanja: ' + low.length)
const GROUPS = [['voznja', 'nakljucje'], ['jedro'], ['izris'], ['zasnova', 'obremenitev']]
const VERDICTS = {
  type: 'object',
  properties: {
    verdicts: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          title: { type: 'string', description: 'naslov najdbe, dobesedno' },
          verdict: { type: 'string', enum: ['potrjeno', 'ovrzeno', 'nejasno'] },
          reason: { type: 'string', description: 'kaj si pognal ali prebral in kaj se je pokazalo' },
          fix: { type: 'string', description: 'najmanjši popravek, če je potrjeno; opozorilo, če predlagani popravek krši pravilo' },
        },
        required: ['title', 'verdict', 'reason'],
      },
    },
  },
  required: ['verdicts'],
}
phase('Preverjanje')
const checked = await parallel(GROUPS.map((g, i) => () => {
  const fs = serious.filter(f => g.includes(f.lens))
  if (!fs.length) return Promise.resolve({ verdicts: [] })
  return agent(COMMON + `\nYour key: skeptik${i}\n\nYou are a SKEPTIC. Other reviewers reported the findings below. For each one try to REFUTE it: reproduce the evidence yourself (read the code at the named place, rerun or write the smallest script; old tree via git worktree under ${SCR}/skeptik${i}/wt if needed). Verdict 'potrjeno' only if you reproduced it or the code unambiguously shows it; 'ovrzeno' if the claim is wrong, already handled elsewhere, or already documented as an accepted deviation; 'nejasno' if you could not decide (say what is missing). Also judge the suggested fix: would it break one of the hard rules (driving invariant, no new rand(), frozen tracks, online untouched)? If so, give a safe alternative.\n\nFindings (JSON):\n` + JSON.stringify(fs, null, 1), { label: 'skeptik:' + g.join('+'), phase: 'Preverjanje', schema: VERDICTS })
}))
const verdicts = checked.filter(Boolean).flatMap(c => c.verdicts)
const byTitle = {}
verdicts.forEach(v => { byTitle[v.title] = v })
return {
  summaries: ok.map(r => ({ key: r.key, summary: r.summary })),
  failedLenses: LENSES.map(l => l.key).filter(k => !ok.some(r => r.key === k)),
  serious: serious.map(f => ({ ...f, check: byTitle[f.title] || null })),
  low,
  unmatchedVerdicts: verdicts.filter(v => !serious.some(f => f.title === v.title)),
}
